<script>
(function(){
  var root=document.querySelector('[data-dpdp]');if(!root)return;
  var stage=root.querySelector('[data-stage]'),shot=root.querySelector('[data-shot]');
  // fotoblokjes onder het flesje
  root.querySelectorAll('.dpdp__thumbs button').forEach(function(b){b.addEventListener('click',function(){
    root.querySelectorAll('.dpdp__thumbs button').forEach(function(x){x.setAttribute('aria-pressed',x===b)});
    if(b.dataset.v==='3d'){stage.classList.remove('photo');return}
    shot.src=b.dataset.src;shot.alt=b.getAttribute('aria-label');stage.classList.add('photo');
  })});
  // in winkelmand: eigen knop, daarna opent de winkelmand van Impulse zelf
  var form=root.querySelector('[data-dpdp-form]');if(!form)return;
  var btn=form.querySelector('[name="add"]'),err=form.querySelector('[data-err]');
  form.addEventListener('submit',function(e){
    var drawer=window.theme&&theme.settings&&theme.settings.cartType==='drawer';if(!drawer)return;
    e.preventDefault();btn.disabled=true;err.hidden=true;
    fetch((window.Shopify&&Shopify.routes?Shopify.routes.root:'/')+'cart/add.js',{method:'POST',headers:{'Accept':'application/json'},body:new FormData(form)})
      .then(function(r){return r.json().then(function(j){return{ok:r.ok,j:j}})})
      .then(function(x){btn.disabled=false;
        if(!x.ok){err.textContent=x.j.description||'Dit kon niet in je winkelmand. Probeer het opnieuw.';err.hidden=false;return}
        document.dispatchEvent(new CustomEvent('ajaxProduct:added',{detail:{product:x.j,addToCartBtn:btn}}));
      }).catch(function(){btn.disabled=false;form.submit()});
  });
})();
</script>

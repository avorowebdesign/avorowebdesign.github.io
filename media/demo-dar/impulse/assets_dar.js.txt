/* dar parfum – eigen onderdelen bovenop Impulse (AVORO). Raakt de code van Impulse niet aan. */
(function(){
  const $=s=>document.querySelector(s),$$=s=>[...document.querySelectorAll(s)];
  const reduce=matchMedia('(prefers-reduced-motion: reduce)').matches;
  const eur=c=>'€'+(c/100).toLocaleString('nl-NL',{minimumFractionDigits:2,maximumFractionDigits:2});

  // winkelmand van Impulse verversen en openen (na toevoegen vanuit een eigen blok)
  const openCart=()=>{document.dispatchEvent(new CustomEvent('cart:build'));setTimeout(()=>document.dispatchEvent(new CustomEvent('cart:open')),250)};
  const addItems=async items=>{const r=await fetch('/cart/add.js',{method:'POST',headers:{'Content-Type':'application/json',Accept:'application/json'},body:JSON.stringify({items})});const j=await r.json();if(!r.ok)throw j;openCart();refreshFs();return j};

  // balk "nog €X tot gratis verzending" in de winkelmand
  const fsEls=$$('[data-dr-fs]');
  const drawFs=c=>fsEls.forEach(el=>{const thr=+el.dataset.threshold||7000;if(!c||!c.item_count){el.innerHTML='';return}
    const left=thr-c.total_price,pct=Math.min(100,Math.round(c.total_price/thr*100));el.classList.toggle('is-done',left<=0);
    el.innerHTML=(left>0?'<p class="dr-fs__t">Nog <b>'+eur(left)+'</b> tot gratis verzending</p>':'<p class="dr-fs__t">Je bestelling wordt <b>gratis verzonden</b></p>')+'<div class="dr-fs__bar" role="progressbar" aria-label="Op weg naar gratis verzending" aria-valuemin="0" aria-valuemax="100" aria-valuenow="'+pct+'"><i style="width:'+pct+'%"></i></div>'});
  // "Nog een geur erbij?" in de winkelmand: max. 2 geuren die er nog niet in zitten, alleen zolang gratis verzending nog niet gehaald is
  const more=$('[data-dr-more]');
  const drawMore=c=>{if(!more)return;const thr=+((fsEls[0]&&fsEls[0].dataset.threshold)||7000);const inCart=new Set((c&&c.items||[]).map(i=>String(i.product_id)));let n=0;
    more.querySelectorAll('[data-pid]').forEach(li=>{const show=n<2&&!inCart.has(li.dataset.pid);li.hidden=!show;if(show)n++});
    more.hidden=!c||!c.item_count||c.total_price>=thr||!n};
  if(more)more.addEventListener('click',async e=>{const b=e.target.closest('[data-dr-add]');if(!b)return;e.preventDefault();b.disabled=true;b.textContent='Toevoegen…';
    try{await fetch('/cart/add.js',{method:'POST',headers:{'Content-Type':'application/json',Accept:'application/json'},body:JSON.stringify({items:[{id:+b.dataset.drAdd,quantity:1}]})});document.dispatchEvent(new CustomEvent('cart:build'));refreshFs()}
    catch(err){}finally{b.disabled=false;b.textContent='Voeg toe'}});
  let fsT=0;const refreshFs=()=>{if(!fsEls.length&&!more)return;clearTimeout(fsT);fsT=setTimeout(async()=>{try{const c=await (await fetch('/cart.js',{headers:{Accept:'application/json'}})).json();drawFs(c);drawMore(c)}catch(e){}},300)};
  if(fsEls.length||more){refreshFs();
    document.addEventListener('ajaxProduct:added',refreshFs);document.addEventListener('cart:build',refreshFs);
    $$('[data-subtotal]').forEach(sub=>new MutationObserver(refreshFs).observe(sub,{childList:true,characterData:true,subtree:true}));}

  // sorteren en filters altijd in het Nederlands, ook als de winkeltaal Engels is
  const NL={'featured':'Uitgelicht','best-selling':'Bestsellers','title-ascending':'Alfabetisch, A-Z','title-descending':'Alfabetisch, Z-A','price-ascending':'Prijs, laag naar hoog','price-descending':'Prijs, hoog naar laag','created-ascending':'Datum, oud naar nieuw','created-descending':'Datum, nieuw naar oud','manual':'Uitgelicht','relevance':'Relevantie'};
  const NLT={'Availability':'Beschikbaarheid','Price':'Prijs','In stock':'Op voorraad','Out of stock':'Niet op voorraad','Product type':'Soort','Brand':'Merk','Vendor':'Merk','Sort by':'Sorteren','Filter':'Filter','Clear all':'Alles wissen','Apply':'Toepassen','Reset':'Wissen'};
  const tr=()=>{$$('select option').forEach(o=>{if(NL[o.value])o.textContent=NL[o.value]});
    $$('.collection-filter__item, .tag-list__header, .collapsible-trigger, .filter-form label, .tag__text, .collection-filter label').forEach(el=>{el.childNodes.forEach(n=>{if(n.nodeType===3){const t=n.textContent.trim();if(NLT[t])n.textContent=n.textContent.replace(t,NLT[t])}})})};
  tr();document.addEventListener('collection:reloaded',tr);document.addEventListener('ajaxProduct:added',tr);

  // geuren-index: het venster rechts toont het flesje van de rij waar je op staat
  const rows=$('#rows'),pv=$('#pvImg');
  if(rows&&pv){
    let cur=rows.querySelector('.is-act');
    const act=li=>{if(!li||li===cur)return;cur&&cur.classList.remove('is-act');li.classList.add('is-act');cur=li;
      const img=new Image();img.src=li.dataset.img;pv.classList.add('is-out');
      const swap=()=>{pv.src=li.dataset.img;$('#pvNo').textContent=li.dataset.no;$('#pvFam').textContent=li.dataset.fam;$('#preview').style.setProperty('--c',getComputedStyle(li).getPropertyValue('--c'));requestAnimationFrame(()=>pv.classList.remove('is-out'))};
      img.complete?setTimeout(swap,180):img.onload=()=>setTimeout(swap,120)};
    rows.addEventListener('pointerover',e=>act(e.target.closest('.dr-rowi')));
    rows.addEventListener('focusin',e=>act(e.target.closest('.dr-rowi')));
  }

  // hero: flesje beweegt licht mee met de muis
  const hb=$('#heroBottle'),hs=$('#hero');
  if(hb&&hs&&!reduce&&matchMedia('(pointer:fine)').matches){let x=0,y=0,tx=0,ty=0;
    hs.addEventListener('pointermove',e=>{const r=hs.getBoundingClientRect();tx=(e.clientX-r.left)/r.width-.5;ty=(e.clientY-r.top)/r.height-.5});
    (function loop(){x+=(tx-x)*.05;y+=(ty-y)*.05;hb.style.setProperty('--px',(x*-18).toFixed(2)+'px');hb.style.setProperty('--py',(y*-12).toFixed(2)+'px');requestAnimationFrame(loop)})();}

  // Dar Fusion: kies 2 of 3 geuren via zoekvelden (nummer, naam, familie, noten, "geïnspireerd op" en zoekwoorden)
  const ff=$('#fusionForm'),fdata=$('#fusionData');
  if(ff&&fdata){
    const ALL=JSON.parse(fdata.textContent).filter(x=>x&&x.no);
    const norm=s=>(s||'').toString().toLowerCase().normalize('NFD').replace(/[̀-ͯ]/g,'').replace(/[^a-z0-9]+/g,' ').trim();
    ALL.forEach(x=>{x.label='No. '+x.no;x.hay=' '+norm([x.label,x.no,x.t,x.f,x.n,x.i,x.k].join(' '))+' '});
    const L=$('#layers'),btn=$('#fusionBtn'),nm=$('#fName'),vn=$('#vname'),mixT=$('#mixText'),price=$('#fPrice');
    const slots=$$('#slots .dr-slot'),TOP=150,BOT=400,NS='http://www.w3.org/2000/svg';
    const pick=[null,null,null];let count=2;
    const need=()=>count-pick.slice(0,count).filter(Boolean).length;
    const draw=()=>{
      const sel=pick.slice(0,count).filter(Boolean);L.innerHTML='';
      const fill=sel.length?Math.min(1,.32*sel.length+.12):0,h=(BOT-TOP)*fill,each=sel.length?h/sel.length:0;
      sel.forEach((x,i)=>{const r=document.createElementNS(NS,'rect');r.setAttribute('x',0);r.setAttribute('width',300);r.setAttribute('y',BOT-each*(i+1));r.setAttribute('height',each+1);r.setAttribute('fill',x.c);r.setAttribute('opacity','.9');L.appendChild(r)});
      if(sel.length){const s=document.createElementNS(NS,'rect');s.setAttribute('x',0);s.setAttribute('width',300);s.setAttribute('y',BOT-h);s.setAttribute('height',3);s.setAttribute('fill','rgba(255,255,255,.35)');L.appendChild(s)}
      mixT.textContent=sel.length?sel.map(x=>x.label).join(' × '):'Kies je eerste geur';
      const n=need();btn.disabled=n>0;btn.textContent=n>0?(n===count?'Kies eerst '+count+' geuren':'Kies nog '+n+(n===1?' geur':' geuren')):'Leg jouw mix in de winkelmand';
      price.textContent=count===3&&ff.dataset.p3?ff.dataset.p3:ff.dataset.p2;
    };
    const setSlot=(i,x)=>{pick[i]=x;const s=slots[i],inp=s.querySelector('input'),dot=s.querySelector('.dr-combo__dot'),cl=s.querySelector('.dr-combo__clear');
      inp.value=x?(x.label+(x.f?', '+x.f:'')):'';dot.style.background=x?x.c:'';dot.classList.toggle('is-set',!!x);cl.hidden=!x;s.classList.toggle('is-set',!!x);draw()};
    const close=s=>{const inp=s.querySelector('input'),ul=s.querySelector('ul');ul.hidden=true;inp.setAttribute('aria-expanded','false');inp.removeAttribute('aria-activedescendant');const i=+s.dataset.slot;if(pick[i])inp.value=pick[i].label+(pick[i].f?', '+pick[i].f:'')};
    const esc=t=>t.replace(/[&<>"]/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;'}[c]));
    const render=(s,q)=>{
      const i=+s.dataset.slot,ul=s.querySelector('ul'),inp=s.querySelector('input'),words=norm(q).split(' ').filter(Boolean);
      const taken=new Set(pick.filter((x,j)=>x&&j!==i).map(x=>x.no));
      let res=ALL.filter(x=>words.every(w=>x.hay.includes(' '+w)||x.hay.includes(w)));
      if(words.length){const w0=words[0].replace(/^0+/,'');res.sort((a,b)=>(String(b.no).replace(/^0+/,'')===w0)-(String(a.no).replace(/^0+/,'')===w0))}
      ul.innerHTML=res.length?res.slice(0,60).map(x=>{const dis=taken.has(x.no);return '<li role="option" id="o'+i+'-'+esc(String(x.no))+'" data-no="'+esc(String(x.no))+'" aria-selected="false"'+(dis?' aria-disabled="true"':'')+'><i style="background:'+esc(x.c)+'"></i><span class="no">'+esc(x.label)+'</span><span class="fam">'+esc(x.f||'')+'</span>'+(x.i?'<span class="nt">Geïnspireerd op '+esc(x.i)+'</span>':'')+(x.n?'<span class="nt">'+esc(x.n)+'</span>':'')+'</li>'}).join('')
        :'<li class="dr-combo__empty" role="presentation">Geen geur gevonden voor “'+esc(q)+'”. <a href="#exclusive">Vraag hem aan via Dar Exclusive</a></li>';
      ul.hidden=false;inp.setAttribute('aria-expanded','true');s._act=-1;
    };
    const move=(s,d)=>{const opts=[...s.querySelectorAll('li[role=option]:not([aria-disabled])')];if(!opts.length)return;s._act=(s._act+d+opts.length)%opts.length;opts.forEach((o,k)=>o.setAttribute('aria-selected',k===s._act));const o=opts[s._act];o.scrollIntoView({block:'nearest'});s.querySelector('input').setAttribute('aria-activedescendant',o.id)};
    const choose=(s,li)=>{if(!li||li.getAttribute('aria-disabled'))return;const x=ALL.find(y=>String(y.no)===li.dataset.no);setSlot(+s.dataset.slot,x);close(s);
      const next=slots.slice(0,count).find((t,k)=>!pick[k]);if(next&&matchMedia('(pointer:fine)').matches)next.querySelector('input').focus()};
    slots.forEach(s=>{const inp=s.querySelector('input'),ul=s.querySelector('ul');
      inp.addEventListener('focus',()=>{inp.select();render(s,pick[+s.dataset.slot]?'':inp.value)});
      inp.addEventListener('input',()=>render(s,inp.value));
      inp.addEventListener('keydown',e=>{if(e.key==='ArrowDown'){e.preventDefault();if(ul.hidden)render(s,inp.value);move(s,1)}else if(e.key==='ArrowUp'){e.preventDefault();move(s,-1)}
        else if(e.key==='Enter'){e.preventDefault();const opts=[...ul.querySelectorAll('li[role=option]:not([aria-disabled])')];choose(s,opts[s._act>=0?s._act:0])}else if(e.key==='Escape'){close(s)}});
      ul.addEventListener('mousedown',e=>e.preventDefault());
      ul.addEventListener('click',e=>choose(s,e.target.closest('li[role=option]')));
      inp.addEventListener('blur',()=>setTimeout(()=>close(s),120));
      s.querySelector('.dr-combo__clear').addEventListener('click',()=>{setSlot(+s.dataset.slot,null);inp.focus()});
    });
    $$('#fusionForm input[name=n]').forEach(r=>r.addEventListener('change',()=>{count=+r.value;slots[2].hidden=count<3;if(count<3&&pick[2])setSlot(2,null);draw()}));
    nm.addEventListener('input',()=>{vn.textContent=nm.value.trim()||'Jouw naam'});
    ff.addEventListener('submit',async e=>{e.preventDefault();if(need()>0)return;const id=+(count===3&&ff.dataset.v3?ff.dataset.v3:ff.dataset.v2);btn.disabled=true;
      try{const props={'Geuren':pick.slice(0,count).map(x=>x.label).join(' + ')};if(nm.value.trim())props['Naam op etiket']=nm.value.trim();await addItems([{id,quantity:1,properties:props}])}
      catch(err){mixT.textContent=(err&&err.description)||'Dat lukte niet, probeer het opnieuw'}finally{btn.disabled=false}});
    // vanaf een productpagina: ?mix=02 zet die geur alvast in vakje 1
    const pre=new URLSearchParams(location.search).get('mix');
    if(pre){const x=ALL.find(y=>String(y.no)===pre);if(x)setSlot(0,x)}
    draw();
  }

  // Geurvinder
  const fd=$('#finderData'),fb=$('#finder');
  if(fd&&fb){
    const D=JSON.parse(fd.textContent),qs=$$('#finder .dr-q'),pg=$$('#fProg li');let step=0,score={};
    const show=i=>{qs.forEach((q,j)=>q.classList.toggle('is-on',j===i));pg.forEach((p,j)=>p.classList.toggle('is-on',j<=Math.min(i,3)))};
    fb.addEventListener('click',e=>{const b=e.target.closest('.dr-q__a button');if(!b)return;b.dataset.s.split(',').forEach(k=>score[k]=(score[k]||0)+1);step++;
      if(step>=4){const best=Object.keys(D).sort((a,b)=>(score[b]||0)-(score[a]||0))[0]||Object.keys(D)[0],d=D[best];
        $('#resNo').textContent='No. '+best;$('#resFam').textContent=d.fam||'';$('#resNotes').textContent=d.notes||'';$('#resLink').href=d.url;fb.style.setProperty('--c',d.c)}
      show(step)});
    $('#resAgain').addEventListener('click',()=>{step=0;score={};fb.style.removeProperty('--c');show(0)});
  }

  // productpagina op mobiel: vaste knop onderin zodra je voorbij de knop van Impulse bent gescrold
  const sa=$('#drSatc'),sb=$('#drSatcBtn');
  const real=$('.product-single__form [name="add"]')||$('form[action*="/cart/add"] [name="add"]');
  if(sa&&sb&&real&&'IntersectionObserver' in window){const mq=matchMedia('(max-width:768px)');let seen=true;
    const upd=()=>{const on=mq.matches&&!seen;sa.classList.toggle('is-on',on);sa.setAttribute('aria-hidden',!on);on?sa.removeAttribute('inert'):sa.setAttribute('inert','');document.body.classList.toggle('dr-satc-on',on);if(on)document.documentElement.style.setProperty('--dr-satc-h',sa.offsetHeight+'px')};
    new IntersectionObserver(([en])=>{seen=en.isIntersecting||en.boundingClientRect.top>0;upd()}).observe(real);
    mq.addEventListener('change',upd);
    sb.addEventListener('click',()=>real.click());
  }
})();
